import json
import math

import rclpy
from geometry_msgs.msg import TransformStamped, Twist
from nav_msgs.msg import Odometry
from rclpy.node import Node
from std_msgs.msg import Float64MultiArray, MultiArrayDimension

try:
    import serial
except ImportError:  # pragma: no cover - handled at runtime on robot
    serial = None


def yaw_to_quaternion(yaw):
    half = yaw * 0.5
    return 0.0, 0.0, math.sin(half), math.cos(half)


class JsonStreamParser:
    """Incrementally parses JSON objects, including several objects on one line."""

    def __init__(self):
        self._decoder = json.JSONDecoder()
        self._buffer = ''

    def feed(self, text):
        self._buffer += text
        objects = []

        while self._buffer:
            self._buffer = self._buffer.lstrip()
            if not self._buffer:
                break

            start = self._buffer.find('{')
            if start < 0:
                self._buffer = ''
                break
            if start > 0:
                self._buffer = self._buffer[start:]

            try:
                obj, index = self._decoder.raw_decode(self._buffer)
            except json.JSONDecodeError:
                break

            objects.append(obj)
            self._buffer = self._buffer[index:]

        return objects


class SerialBridge(Node):
    def __init__(self):
        super().__init__('serial_bridge')
        
        self.declare_parameter('port', 'COM3')
        self.declare_parameter('baud', 115200)
        self.declare_parameter('odom_frame_id', 'odom')
        self.declare_parameter('base_frame_id', 'base_link')
        self.declare_parameter('ticks_per_meter', 2690.0)
        self.declare_parameter('publish_tf', False)
        self.declare_parameter('command_format', 'json')
        self.declare_parameter('cmd_vel_timeout', 0.5)
        self.declare_parameter('command_rate_hz', 10.0)

        self.port = self.get_parameter('port').value
        self.baud = int(self.get_parameter('baud').value)
        self.odom_frame_id = self.get_parameter('odom_frame_id').value
        self.base_frame_id = self.get_parameter('base_frame_id').value
        self.ticks_per_meter = float(self.get_parameter('ticks_per_meter').value)
        self.publish_tf = bool(self.get_parameter('publish_tf').value)
        self.command_format = self.get_parameter('command_format').value
        self.cmd_vel_timeout = float(self.get_parameter('cmd_vel_timeout').value) # the important parameter

        self.serial = self._open_serial()
        self.parser = JsonStreamParser()
        self.last_cmd = Twist()
        self.last_cmd_time = self.get_clock().now()

        self.encoder_pub = self.create_publisher(Float64MultiArray, '/encoder_ticks', 10)
        self.odom_pub = self.create_publisher(Odometry, '/odom_raw', 10)
        self.cmd_sub = self.create_subscription(Twist, '/cmd_vel', self._cmd_callback, 10)

        if self.publish_tf:
            from tf2_ros import TransformBroadcaster

            self.tf_broadcaster = TransformBroadcaster(self)
        else:
            self.tf_broadcaster = None

        self.read_timer = self.create_timer(0.01, self._read_serial)
        command_period = 1.0 / max(1.0, float(self.get_parameter('command_rate_hz').value))
        self.command_timer = self.create_timer(command_period, self._write_command)

        self.serial.write(b"0.0000,0.0000\n")

        self.get_logger().info(
            f'Listening on {self.port} at {self.baud} baud, ticks_per_meter={self.ticks_per_meter:.1f}'
        )

    def _open_serial(self):
        if serial is None:
            raise RuntimeError(
            'pyserial is required.'
        )

        ser = serial.Serial()

        ser.port = self.port
        ser.baudrate = self.baud

        ser.dtr = False
        ser.rts = False

        ser.open()

        return ser  

   

    def _cmd_callback(self, msg):
        self.get_logger().info(
            f'CMD RX -> linear={msg.linear.x:.3f} angular={msg.angular.z:.3f}'
        )

        self.last_cmd = msg
        self.last_cmd_time = self.get_clock().now()

    def _read_serial(self):
        try:
            waiting = self.serial.in_waiting
            if waiting <= 0:
                return
            text = self.serial.read(waiting).decode('utf-8', errors='ignore')
        except Exception as exc:
            self.get_logger().warn(f'Failed reading serial data: {exc}')
            return

        for obj in self.parser.feed(text):
            if obj.get('type') == 'debug':
                self.get_logger().info(
                    f"ARDUINO RX -> linear={obj['linear']} angular={obj['angular']}"
                )

            if obj.get('type') == 'odom':
                self._publish_odom(obj)

    def _publish_odom(self, obj):
        now = self.get_clock().now().to_msg()

        left_ticks = float(obj.get('left_ticks', 0.0))
        right_ticks = float(obj.get('right_ticks', 0.0))
        avg_ticks_abs = float(obj.get('avg_ticks_abs', (abs(left_ticks) + abs(right_ticks)) * 0.5))

        ticks_msg = Float64MultiArray()
        ticks_msg.layout.dim = [MultiArrayDimension(label='left_right_avg_abs', size=3, stride=3)]
        ticks_msg.data = [left_ticks, right_ticks, avg_ticks_abs]
        self.encoder_pub.publish(ticks_msg)

        x = float(obj.get('x', 0.0))
        y = float(obj.get('y', 0.0))
        theta = float(obj.get('theta', 0.0))
        v = float(obj.get('v', 0.0))
        omega = float(obj.get('omega', 0.0))

        odom = Odometry()
        odom.header.stamp = now
        odom.header.frame_id = self.odom_frame_id
        odom.child_frame_id = self.base_frame_id
        odom.pose.pose.position.x = x
        odom.pose.pose.position.y = y
        qx, qy, qz, qw = yaw_to_quaternion(theta)
        odom.pose.pose.orientation.x = qx
        odom.pose.pose.orientation.y = qy
        odom.pose.pose.orientation.z = qz
        odom.pose.pose.orientation.w = qw
        odom.twist.twist.linear.x = v
        odom.twist.twist.angular.z = omega
        odom.pose.covariance[0] = 0.02
        odom.pose.covariance[7] = 0.02
        odom.pose.covariance[35] = 0.05
        odom.twist.covariance[0] = 0.02
        odom.twist.covariance[35] = 0.05
        self.odom_pub.publish(odom)

        if self.tf_broadcaster is not None:
            transform = TransformStamped()
            transform.header.stamp = now
            transform.header.frame_id = self.odom_frame_id
            transform.child_frame_id = self.base_frame_id
            transform.transform.translation.x = x
            transform.transform.translation.y = y
            transform.transform.rotation = odom.pose.pose.orientation
            self.tf_broadcaster.sendTransform(transform)

    def _write_command(self):
        command = self.last_cmd

        age = (self.get_clock().now() - self.last_cmd_time).nanoseconds * 1e-9

        if age > self.cmd_vel_timeout:
            command = Twist()

        linear = float(command.linear.x)
        angular = float(command.angular.z)

        payload = f'{linear:.4f},{angular:.4f}\n'

        self.get_logger().info(
            f'TX -> {payload.strip()}'
        )

        try:
            self.serial.write(payload.encode('utf-8'))
        except Exception as exc:
            self.get_logger().warn(
                f'Failed writing serial command: {exc}'
            )


def main(args=None):
    rclpy.init(args=args)

    node = SerialBridge()

    try:
        rclpy.spin(node)

    except KeyboardInterrupt:
        pass

    finally:
        node.destroy_node()

        if rclpy.ok():
            rclpy.shutdown()


if __name__ == '__main__':
    main()

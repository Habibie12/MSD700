void receivePropoSwitch() {

  static int lastRelayState = 0;
  static int relayStableCounter = 0;

  static int lastModeState = 1;        // default: RC control
  static int modeStableCounter = 0;

  int toggle_A = pulseIn(ch5, HIGH, 25000);  // Power relay
  int toggle_B = pulseIn(ch6, HIGH, 25000);  // Mode select (future Mini PC)

  // ---------------------------
  // Validate signals first
  // ---------------------------
  if (toggle_A < 900 || toggle_A > 2100) {
    return;  // ignore invalid signal
  }

  if (toggle_B < 900 || toggle_B > 2100) {
    return;  // ignore invalid signal
  }

  int newRelayState; // Relay state: 0 = OFF, 1 = ON

  if (toggle_A < 1400) {
    newRelayState = 0;   // OFF
  }
  else if (toggle_A > 1600) {
    newRelayState = 1;   // ON
  }
  else {
    newRelayState = lastRelayState;  // deadband
  }

  if (newRelayState == lastRelayState) {
    relayStableCounter++;
  } else {
    relayStableCounter = 0;
  }

  if (relayStableCounter > 5) {
    g_power_relay = newRelayState;
    lastRelayState = newRelayState;
  }


  int newModeState; // mini PC mode: 0, RC mode: 1

  if (toggle_B < 1400) {
    newModeState = 0;   // Mini PC mode
  }
  else if (toggle_B > 1600) {
    newModeState = 1;   // RC mode
  }
  else {
    newModeState = lastModeState;
  }

  if (newModeState == lastModeState) {
    modeStableCounter++;
  } else {
    modeStableCounter = 0;
  }

  if (modeStableCounter > 5) {
    g_select_controler = newModeState;
    lastModeState = newModeState;
  }
}


void receivePropoJoy(float deadzone) {  // input proportional controler joysticks
  int PROPO_JOY_MAX = 1918;
  int PROPO_JOY_MIN = 1104;
  int rawY = pulseIn(ch2, HIGH, 25000);
  int rawX = pulseIn(ch4, HIGH, 25000);

  if(rawY < 900 || rawY > 2100 || rawX < 900 || rawX > 2100){
    return;  // invalid signal → ignore
  }

float input_propo_joy_leftY = map(rawY, PROPO_JOY_MAX, PROPO_JOY_MIN, -100, 100) / 100.0;
float input_propo_joy_leftX = map(rawX, PROPO_JOY_MAX, PROPO_JOY_MIN, -100, 100) / 100.0;

input_propo_joy_leftY = constrain(input_propo_joy_leftY, -1.0, 1.0);
input_propo_joy_leftX = constrain(input_propo_joy_leftX, -1.0, 1.0);

// dead zone
  if (-deadzone <= input_propo_joy_leftX && input_propo_joy_leftX <= deadzone) {
    input_propo_joy_leftX = 0.00;
  }
  if (-deadzone <= input_propo_joy_leftY && input_propo_joy_leftY <= deadzone) {
    input_propo_joy_leftY = 0.00;
  }

  if (g_mode_switch == 0) {  // フタが後ろ | lid back
    g_input_propo_joy_leftY = input_propo_joy_leftY;
  } else if (g_mode_switch == 1) {  // フタが前 | lid front
    g_input_propo_joy_leftY = -input_propo_joy_leftY;
  }
  g_input_propo_joy_leftX = input_propo_joy_leftX;

  Serial.print(" g_input_propo_joy_leftY ");
  Serial.print(g_input_propo_joy_leftY);
  Serial.print(" g_input_propo_joy_leftX ");
  Serial.println(g_input_propo_joy_leftX);
}

float velToRpm(float v) {
  // convert velocity to rpm
  const float REDUCTION_RATIO = 100.0;  // 減速比 reduction ratio 
  const float WHEEL_RADIUS = 0.1105;    // 車輪半径 radius of gyration [m]
  float n = 30 / M_PI * REDUCTION_RATIO / WHEEL_RADIUS * v;

  return n;
}

void ik(float v, float w) {
  // calculate linear velocity and angur velocity to left and right velocity
  float WHEEL_DISTANCE = 0.600;  // 車輪間距離 distance between the wheels [m]
  float v_l = v - WHEEL_DISTANCE * w;
  float v_r = v + WHEEL_DISTANCE * w;

  float left_rpm = velToRpm(v_l);
  float right_rpm = velToRpm(v_r);

  // direction of motor
  if (left_rpm > 0) {
    digitalWrite(mLeftFWD, HIGH);
    digitalWrite(mLeftREV, LOW);
    // Serial.print("Left dir:FWD, ");
  } else if (left_rpm < 0) {
    digitalWrite(mLeftFWD, LOW);
    digitalWrite(mLeftREV, HIGH);
    // Serial.print("Left dir:REV, ");
  } else {
    digitalWrite(mLeftFWD, LOW);
    digitalWrite(mLeftREV, LOW);
    // Serial.print("Left dir:stop, ");
  }

  if (right_rpm > 0) {
    digitalWrite(mRightFWD, LOW);
    digitalWrite(mRightREV, HIGH);
    // Serial.print("Right dir:FWD, ");
  } else if (right_rpm < 0) {
    digitalWrite(mRightFWD, HIGH);
    digitalWrite(mRightREV, LOW);
    // Serial.print("Right dir:REV, ");
  } else {
    digitalWrite(mRightFWD, LOW);
    digitalWrite(mRightREV, LOW);
    // Serial.print("Right dir:stop, ");
  }

  //  0: フタが後ろ 1: フタが前 | 0:lid back 1:lid front
  if (g_mode_switch == 0) {  // フタが後ろ | lid back
    if (left_rpm < 0 || right_rpm < 0) {
      g_left_rpm = right_rpm;
      g_right_rpm = left_rpm;
    } else {
      g_left_rpm = left_rpm;
      g_right_rpm = right_rpm;
    }
  } else if (g_mode_switch == 1) {  // フタが前 | lid front
    if (left_rpm < 0 || right_rpm < 0) {
      g_left_rpm = left_rpm;
      g_right_rpm = right_rpm;
    } else {
      g_left_rpm = right_rpm;
      g_right_rpm = left_rpm;
    }
  }
}

void stopMotor(bool motor_brake_state) {
  // HIGH:ブレーキ解放 LOW:ブレーキ保持 | HIGH:brake release LOW:brake hold
  if (motor_brake_state == 0) {
    digitalWrite(mbFreeL, HIGH);
    digitalWrite(mbFreeR, HIGH);
  } else if (motor_brake_state == 1) {
    digitalWrite(mbFreeL, LOW);
    digitalWrite(mbFreeR, LOW);
  }
  analogWrite(mmLPin, 0);
  analogWrite(mmRPin, 0);
  //=== STOP表示 ===
  // Serial.print("Motor stop, ");
}

void writeMotorPwm(int left_rpm, int right_rpm) {
  int left_pwm;
  int right_pwm;

  left_rpm = abs(left_rpm);
  right_rpm = abs(right_rpm);

  left_rpm = min(left_rpm, 4000);
  left_rpm = max(left_rpm, 0);

  right_rpm = min(right_rpm, 4000);
  right_rpm = max(right_rpm, 0);

  left_pwm = map(left_rpm, 0, 4000, 0, 255);
  right_pwm = map(right_rpm, 0, 4000, 0, 255);

  if (left_pwm && right_pwm == 0) {
    stopMotor(g_motor_brake);
  } else {
    analogWrite(mmLPin, left_pwm);
    analogWrite(mmRPin, right_pwm);
    // Serial.print("left_pwm: ");
    // Serial.print(left_pwm);
    // Serial.print(" , ");
    // Serial.print("right_pwm: ");
    // Serial.println(right_pwm);
  }
}

//アームリフト関数 | Arm lift function
void lift(int pulse) {
  if (pulse < 1250 && pulse > 950) {
    digitalWrite(liftFWD, HIGH);
    digitalWrite(liftREV, LOW);
  } else if (pulse < 2070 && pulse > 1750) {
    digitalWrite(liftFWD, LOW);
    digitalWrite(liftREV, HIGH);
  } else if (pulse < 1750 && pulse > 1250) {
    digitalWrite(liftFWD, LOW);
    digitalWrite(liftREV, LOW);
  }
}

//アームダンプ関数 | Arm dump function
void dump(int pulse) {
  if (pulse < 1250 && pulse > 950) {
    digitalWrite(dumpFWD, HIGH);
    digitalWrite(dumpREV, LOW);
  } else if (pulse < 2070 && pulse > 1750) {
    digitalWrite(dumpFWD, LOW);
    digitalWrite(dumpREV, HIGH);
  } else if (pulse < 1750 && pulse > 1250) {
    digitalWrite(dumpFWD, LOW);
    digitalWrite(dumpREV, LOW);
  }
}

float readBatteryVoltage() {

  long sum = 0;

  for (int i = 0; i < 20; i++) {
    sum += analogRead(BatteryPin);
    delay(2);
  }

  float adcValue = sum / 20.0;

  float Vref = 4.1; //adjust these values based on reference voltage of buck converter
  float vA6 = adcValue * (Vref / 1023.0);
  float dividerRatio = (51.0 + 9.1) / 9.1;
  float batteryVoltage = vA6 * dividerRatio  * 1.031; // 1.031 is a calibration factor to match actual voltage readings

  return batteryVoltage;
}

void checkBatteryStatus() {

  float batteryVoltage = readBatteryVoltage();

  Serial.print(" Volt: ");
  Serial.println(batteryVoltage, 2);  // 小数点以下2桁で表示 | Display with 2 decimal places

  // 電圧が20V未満の場合、走行モーターのみ稼働 | If voltage is below 20V, only driving motors are active
  if (batteryVoltage < 20.00 && batteryVoltage >= 18.0) {
    Serial.println("Voltage below 20V. Only driving motors are active.");
    digitalWrite(LightRed, HIGH);
    digitalWrite(liftFWD, LOW);
    digitalWrite(liftREV, LOW);
    digitalWrite(dumpFWD, LOW);
    digitalWrite(dumpREV, LOW);
  } else if (batteryVoltage < 18.0) {
    Serial.println("Voltage below 18V. Shutting down all operations.");
    digitalWrite(LightRed, HIGH);
    stopMotor(g_motor_brake);  // すべての動作を停止 | Stop all operations
    digitalWrite(liftFWD, LOW);
    digitalWrite(liftREV, LOW);
    digitalWrite(dumpFWD, LOW);
    digitalWrite(dumpREV, LOW);
  } else  // volt >= 20.0 のとき
  {
    Serial.println("Voltage normal. All operations are allowed.");
    digitalWrite(LightRed, LOW);  // 警告LEDを消す | Turn off warning LED
  }
}

void displaySOC(int soc, float voltage) {

  lcd.setCursor(0,0);
  lcd.print("BATTERY STATUS   ");

  lcd.setCursor(0,1);
  lcd.print("Volt: ");
  lcd.print(voltage,1);
  lcd.print("V     ");

  lcd.setCursor(0,2);
  lcd.print("SOC : ");
  lcd.print(soc);
  lcd.print("%     ");

  lcd.setCursor(0,3);
  lcd.print("[");
  
  int totalBars = 16;   // cleaner than 20
  int filledBars = map(soc, 0, 100, 0, totalBars);

  for(int i=0; i<totalBars; i++){
    if(i < filledBars)
      lcd.write(255);   // full block
    else
      lcd.print(" ");
  }

  lcd.print("]");
}


// ------------------- SOC function meassurement -------------------
int calculateSOC(float voltage) {

  if (voltage >= 29.4) return 100;
  if (voltage <= 18.0) return 0;

  if (voltage >= 27.5) {
    // 80–100%
    return 80 + (voltage - 27.5) * (20.0 / (29.4 - 27.5));
  }
  else if (voltage >= 24.5) {
    // 40–80%
    return 40 + (voltage - 24.5) * (40.0 / (27.5 - 24.5));
  }
  else {
    // 0–40%
    return (voltage - 18.0) * (40.0 / (24.5 - 18.0));
  }
}



void setup() {

delay(2000);

Serial.begin(9600);

bool rcReady = false;

unsigned long startTime = millis();

while(!rcReady){
  int pulse = pulseIn(ch2, HIGH, 25000);
  if(pulse > 900 && pulse < 2100){
        rcReady = true;
    }

  if(millis() - startTime > 5000){
        break;  // continue even if no RC
    }
}

  //モータードライバー速度設定 | Motor driver speed setting pins
  pinMode(mmLPin, OUTPUT);
  pinMode(mmRPin, OUTPUT);

  //左モータードライバーアウトプット | Left motor driver output
  pinMode(mLeftFWD, OUTPUT);
  pinMode(mLeftREV, OUTPUT);
  pinMode(stopModeL, OUTPUT);
  pinMode(m0L, OUTPUT);
  pinMode(mbFreeL, OUTPUT);

  //右モータードライバーアウトプット | Right motor driver output
  pinMode(mRightFWD, OUTPUT);
  pinMode(mRightREV, OUTPUT);
  pinMode(stopModeR, OUTPUT);
  pinMode(m0R, OUTPUT);
  pinMode(mbFreeR, OUTPUT);

  //動力リレー用リレー | Power relay control
  pinMode(powerRelay, OUTPUT);

  // HIGH:時間設定 LOW:瞬時停止 | HIGH: timed stop LOW: immediate stop
  digitalWrite(stopModeR, LOW);
  digitalWrite(stopModeL, LOW);

  // HIGH:速度指定を外部入力 LOW:内部速度設定器 | HIGH: external speed control LOW: internal speed controller
  digitalWrite(m0R, HIGH);
  digitalWrite(m0L, HIGH);

  //HIGH:ブレーキ解放 LOW:ブレーキ保持 | HIGH:brake release LOW:brake hold
  digitalWrite(mbFreeR, HIGH);
  digitalWrite(mbFreeL, HIGH);

  //シリンダーモータードライバー | Cylinder / Actuator driver pins
  pinMode(liftFWD, OUTPUT);
  pinMode(liftREV, OUTPUT);
  pinMode(dumpFWD, OUTPUT);
  pinMode(dumpREV, OUTPUT);

  //レシーバー | RC receiver pins
  pinMode(ch1, INPUT);
  pinMode(ch2, INPUT);
  pinMode(ch3, INPUT);
  pinMode(ch4, INPUT);
  pinMode(ch5, INPUT);
  pinMode(ch6, INPUT);
  pinMode(ch7, INPUT);
  pinMode(ch8, INPUT);

  //積層信号灯用リレー | Stack light relay
  pinMode(LightRed, OUTPUT);

  //積層信号灯用リレー 
  pinMode(BatteryPin, INPUT);

  filteredVolt = readBatteryVoltage();
  for(int i=0;i<8;i++){
  lcd.createChar(i, bigChars[i]);
}

lcd.init();
lcd.backlight();
lcd.clear();
}


void loop() {

  receivePropoSwitch();
  receivePropoJoy(0.15);

  int testPulse = pulseIn(ch2, HIGH, 25000);

  if(testPulse > 900 && testPulse < 2100){
    lastSignalTime = millis();
  }

  // If no signal for 200ms → stop everything
  if(millis() - lastSignalTime > 200){
    digitalWrite(powerRelay, LOW);
    stopMotor(g_motor_brake);
    return;  // skip rest of loop
  }

  g_linear_x = g_input_propo_joy_leftY * LINEAR_SPEED_MAX;
  g_angular_z = g_input_propo_joy_leftX * ANGULAR_SPEED_MAX;

  volt = readBatteryVoltage();
  checkBatteryStatus();  // 電圧チェック（安全制御・LED点滅）| Voltage check (safety control / LED blinking)
  delay(10);

  switch (g_power_relay) {
    case 0:
      digitalWrite(powerRelay, LOW);
      stopMotor(g_motor_brake);
      break;
    case 1:
      if (volt >= 18.0) {
        digitalWrite(powerRelay, HIGH);
        ik(g_linear_x, g_angular_z);
        writeMotorPwm(g_left_rpm, g_right_rpm);
        rightX = pulseIn(ch1, HIGH, 25000);
        rightY = pulseIn(ch3, HIGH, 25000);
        lift(rightY);
        dump(rightX);
        break;
      } else {
        // 電源スイッチOFFまたは電圧不足時：停止 | Power switch OFF or insufficient voltage: stop
        digitalWrite(powerRelay, LOW);
        stopMotor(g_motor_brake);
      }
  }


float alpha = 0.85;
filteredVolt = alpha * filteredVolt + (1.0 - alpha) * volt;
int soc = calculateSOC(filteredVolt);
Serial.print("Battery Voltage remains: ");
Serial.print(soc);

static unsigned long lastUpdate = 0;

if(millis() - lastUpdate > 300){
    displaySOC(soc, filteredVolt);
    lastUpdate = millis();
}
